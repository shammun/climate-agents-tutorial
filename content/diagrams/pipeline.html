Core analysis snippet

import xarray as xr, scipy.stats as st
ds = xr.open_mfdataset("data/raw/era5/*.nc")
jja = ds.t2m.sel(time=ds.time.dt.month.isin([6,7,8]))
mean = jja.groupby("time.year").mean("time")
